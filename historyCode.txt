Firebase 보안 규칙 기록
- 이번 단계는 Firebase Authentication과 Firestore users 컬렉션 생성 기능을 추가했으므로 테스트 가능한 Firestore 보안 규칙 예시를 함께 기록한다.
- 아래 규칙은 사용자가 본인 users 문서만 생성/조회/수정할 수 있고, 회원가입 시 role은 user로만 생성되도록 제한한다.
- 관리자 기능은 아직 구현하지 않았으므로 admin 판정은 다음 관리자 단계에서 별도 보강해야 한다.

Firestore rules 예시:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    match /users/{uid} {
      allow create: if isOwner(uid)
        && request.resource.data.keys().hasOnly(['email', 'nickname', 'role', 'createAt'])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.role == 'user'
        && request.resource.data.createAt == request.time;

      allow read: if isOwner(uid);

      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).changedKeys().hasOnly(['nickname'])
        && request.resource.data.role == resource.data.role
        && request.resource.data.email == resource.data.email;

      allow delete: if false;
    }
  }
}

테스트 기준
- 회원가입한 사용자는 users/{본인 uid} 문서를 role user로 생성할 수 있어야 한다.
- 클라이언트에서 role을 admin 등 다른 값으로 생성하려 하면 거부되어야 한다.
- 로그인한 사용자는 본인 users 문서만 읽을 수 있어야 한다.
- 다른 사용자의 users 문서 읽기/수정/삭제는 거부되어야 한다.

Firestore rules 예시:

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    match /users/{uid} {
      allow create: if isOwner(uid)
        && request.resource.data.keys().hasOnly(['email', 'nickname', 'role', 'createAt'])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.role == 'user'
        && request.resource.data.createAt == request.time;

      allow read: if isOwner(uid);

      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).changedKeys().hasOnly(['nickname'])
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.email == resource.data.email
        && request.resource.data.role == resource.data.role
        && request.resource.data.createAt == resource.data.createAt;

      allow delete: if false;
    }
  }
}

        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.role == 'user'
        && request.resource.data.createAt == request.time;

      allow read: if isOwner(uid);

      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).changedKeys().hasOnly(['nickname'])
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.email == resource.data.email
        && request.resource.data.role == resource.data.role
        && request.resource.data.createAt == resource.data.createAt;

      allow delete: if false;
    }

    match /orders/{orderId} {
      allow create: if isSignedIn()
        && request.resource.data.keys().hasOnly([
          'userId', 'uid', 'email', 'items', 'subtotal', 'deliveryfree',
          'totalPrice', 'createAt', 'status'
        ])
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.subtotal is number
        && request.resource.data.deliveryfree is number
        && request.resource.data.totalPrice is number
        && request.resource.data.totalPrice >= request.resource.data.subtotal
        && request.resource.data.createAt == request.time
        && request.resource.data.status == 'ordered';

      allow read: if isSignedIn()
        && (
          resource.data.userId == request.auth.uid ||
          resource.data.uid == request.auth.uid
        );

      allow update, delete: if false;
    }
  }
}

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }

    function isOwner(uid) {
      return isSignedIn() && request.auth.uid == uid;
    }

    function isAdmin() {
      return isSignedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isInitialAdminEmail() {
      return isSignedIn() && request.auth.token.email == 'admin@a.com';
    }

    function hasOwner() {
      return isSignedIn()
        && (
          resource.data.userId == request.auth.uid ||
          resource.data.uid == request.auth.uid
        );
    }

    function requestHasOwner() {
      return isSignedIn()
        && (
          request.resource.data.userId == request.auth.uid ||
          request.resource.data.uid == request.auth.uid
        );
    }

    match /users/{uid} {
      allow create: if isOwner(uid)
        && request.resource.data.keys().hasOnly(['email', 'nickname', 'role', 'createAt'])
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.role == 'user'
        && request.resource.data.createAt == request.time;

      allow get: if isOwner(uid) || isAdmin();
      allow list: if isAdmin();

      allow update: if isOwner(uid)
        && request.resource.data.diff(resource.data).changedKeys().hasOnly(['nickname'])
        && request.resource.data.nickname is string
        && request.resource.data.nickname.size() > 0
        && request.resource.data.email == resource.data.email
        && request.resource.data.role == resource.data.role
        && request.resource.data.createAt == resource.data.createAt;

      allow delete: if false;

      match /{document=**} {
        allow read, write: if isOwner(uid);
      }
    }

    match /orders/{orderId} {
      allow create: if isSignedIn()
        && request.resource.data.keys().hasOnly([
          'userId', 'uid', 'email', 'items', 'subtotal', 'deliveryfree',
          'totalPrice', 'createAt', 'status'
        ])
        && request.resource.data.userId == request.auth.uid
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.items is list
        && request.resource.data.items.size() > 0
        && request.resource.data.subtotal is number
        && request.resource.data.deliveryfree is number
        && request.resource.data.totalPrice is number
        && request.resource.data.totalPrice >= request.resource.data.subtotal
        && request.resource.data.createAt == request.time
        && request.resource.data.status == 'ordered';

      allow read: if hasOwner() || isAdmin();

      allow update, delete: if false;
    }

        ])

        ])

        && request.resource.data.uid == resource.data.uid;
